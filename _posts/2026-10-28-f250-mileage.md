I've been keeping a spreadsheet of fillup data on my 2003 F-250 for several years now. Below are a few interesting charts visualizing that data.

### Fuel Economy Over Time

This chart shows the average fuel economy (in miles per gallon) of my 2003 Ford F-250 over time, based on odometer readings. Each point represents the average MPG for a 1,000-mile bucket of mileage, with error bars indicating the standard error of the mean for that bucket.


<svg id="chart" viewBox="0 0 600 400"></svg>

<script>
const drawChart1 = async () => {
    // -------------------------------
    // Chart 1: Fuel Economy Over Time (Error Bar Chart)
    // -------------------------------
    d3.csv(
        "https://docs.google.com/spreadsheets/d/e/2PACX-1vQKOSZP29z4OJA327iSPq4e7pfPF8asl6Yjvzw_pUNWHLDrEttYbub2yHpGEDTyylmaYnUpW0mJXDDf/pub?gid=1357971635&single=true&output=csv"
    ).then(rawData => {
        // Clean and transform raw logs
        const cleanData = rawData
            .filter(d => d.Mileage && d.MPG && !isNaN(d.MPG) && d.Year !== "1899")
            .map(d => ({
                mileage: +d.Mileage.replace(/,/g, ""),
                mpg: +d.MPG
            }))
            .filter(d => d.mileage > 0 && d.mpg > 0);

        // Aggregate Data into 1,000-Mile Buckets
        const bucketSize = 2500;
        const nestedData = d3.nest()
            .key(d => Math.floor(d.mileage / bucketSize) * bucketSize)
            .rollup(values => {
                const mpgArray = values.map(v => v.mpg);
                const mean = d3.mean(mpgArray);
                const deviation = d3.deviation(mpgArray) || 0;
                // Standard Error of the Mean (SEM) = StdDev / sqrt(N)
                const sem = deviation / Math.sqrt(mpgArray.length); 
                
                return {
                    bucketCenter: +values[0].mileage, // Use exact center or structural key
                    mean: mean,
                    upperBound: mean + (sem || 0.5), // Fallback cushion if bucket has 1 element
                    lowerBound: Math.max(0, mean - (sem || 0.5))
                };
            })
            .entries(cleanData);

        const bucketData = nestedData.map(d => ({
            mileageKey: +d.key + (bucketSize / 2), // Position center point in middle of the 1k window
            ...d.value
        })).sort((a, b) => a.mileageKey - b.mileageKey);

        // Chart Layout Setup
        const margin = { top: 50, right: 30, bottom: 50, left: 60 };
        const width = 600 - margin.left - margin.right;
        const height = 400 - margin.top - margin.bottom;

        const svg = d3.select("#chart")
            .append("svg")
            .attr("width", width + margin.left + margin.right)
            .attr("height", height + margin.top + margin.bottom)
            .append("g")
            .attr("transform", `translate(${margin.left},${margin.top})`);

        // Mathematical Scales
        const xScale = d3.scaleLinear()
            .domain([
            d3.min(bucketData, d => d.mileageKey) - 1000,
            d3.max(bucketData, d => d.mileageKey) + 1000
            ])
            .range([0, width]);

        const yScale = d3.scaleLinear()
            .domain([
            d3.min(bucketData, d => d.lowerBound) * 0.9,
            d3.max(bucketData, d => d.upperBound) * 1.1
            ])
            .range([height, 0]);
        
        // Key dates that show up as annotated vertical rules
        const keyEvents = [
            { mileage: 183450, label: "Mud tires, steel/bumpers installed" },
            { mileage: 165600, label: "Performance exhaust/kit installed" }
        ];

        svg.selectAll(".important-line")
          .data(keyEvents)
          .enter()
          .append("line")
          .attr("class", "important-line")
          .attr("x1", d => xScale(d.mileage))
          .attr("x2", d => xScale(d.mileage))
          .attr("y1", 0)
          .attr("y2", height)
          .attr("stroke", "red")
          .attr("stroke-dasharray", "4 4") // Dotted or dashed style
          .attr("stroke-width", 1);

        svg.selectAll(".important-label")
          .data(keyEvents)
          .enter()
          .append("text")
          .attr("class", "important-label")
          .style("font-size", "12px")
          .attr("fill", "red")
          .attr("x", d => xScale(d.mileage) + 5) // Slight offset right
          .attr("y", 15) // Near top of the chart
          .selectAll("tspan")
          .data(d => d.label.split("/")) // Split string into an array of lines
          .enter()
          .append("tspan")
          .text(d => d)
          // .attr("dx", 50) // Keeps them horizontally aligned
          .attr("dx", (d, i) => i === 0 ? 0 : -80) // Slight offset right
          .attr("dy", (d, i) => i === 0 ? 0 : "1.2em"); // Offsets the second line down

        // Draw Axes
        svg.append("g")
            .attr("transform", `translate(0,${height})`)
            .call(d3.axisBottom(xScale).tickFormat(d3.format(",")))
            .append("text")
            .attr("x", width / 2)
            .attr("y", 40)
            .attr("fill", "#000")
            .style("text-anchor", "middle")
            .style("font-weight", "bold")
            .text("Odometer Mileage (2,500 Mile Buckets)");

        svg.append("g")
            .call(d3.axisLeft(yScale))
            .append("text")
            .attr("transform", "rotate(-90)")
            .attr("x", -height / 2)
            .attr("y", -40)
            .attr("fill", "#000")
            .style("text-anchor", "middle")
            .style("font-weight", "bold")
            .text("Average Fuel Economy (MPG)");

        // Y-axis gridlines
        svg.append("g")
            .attr("class", "grid")
            .call(d3.axisLeft(yScale)
                .tickSize(-width, 0, 0)
                .tickFormat("")
            );

        // Draw Error Bars (Vertical Core Spindle)
        const errorBars = svg.selectAll(".error-bar")
            .data(bucketData)
            .enter()
            .append("g")
            .attr("class", "error-bar");

        // Main Vertical Range Line
        errorBars.append("line")
            .attr("x1", d => xScale(d.mileageKey))
            .attr("x2", d => xScale(d.mileageKey))
            .attr("y1", d => yScale(d.lowerBound))
            .attr("y2", d => yScale(d.upperBound))
            .attr("stroke", "#777777")
            .attr("stroke-width", 1);

        // Top horizontal cap
        errorBars.append("line")
            .attr("x1", d => xScale(d.mileageKey) - 6)
            .attr("x2", d => xScale(d.mileageKey) + 6)
            .attr("y1", d => yScale(d.upperBound))
            .attr("y2", d => yScale(d.upperBound))
            .attr("stroke", "#777777")
            .attr("stroke-width", 2);

        // Bottom horizontal cap
        errorBars.append("line")
            .attr("x1", d => xScale(d.mileageKey) - 6)
            .attr("x2", d => xScale(d.mileageKey) + 6)
            .attr("y1", d => yScale(d.lowerBound))
            .attr("y2", d => yScale(d.lowerBound))
            .attr("stroke", "#777777")
            .attr("stroke-width", 2);

        // Draw Averaged Data Series Points (The Mean Anchor)
        svg.selectAll(".mean-dot")
            .data(bucketData)
            .enter()
            .append("circle")
            .attr("class", "mean-dot")
            .attr("cx", d => xScale(d.mileageKey))
            .attr("cy", d => yScale(d.mean))
            .attr("r", 4)
            .attr("fill", "#cccccc")
            .attr("stroke", "#777777")
            .attr("stroke-width", 1);
    });
};

drawChart1();
</script>

### Diesel fuel price over time

This chart shows the price of diesel fuel over time, primarily in Texas. Based on data collected from each fillup.


<svg id="dieselprice" viewBox="0 0 600 400"></svg>

<script>
const drawChart2 = async () => {
    // --------------------------------
    // Chart 2: Mileage Histogram
    // --------------------------------
    d3.csv(
        "https://docs.google.com/spreadsheets/d/e/2PACX-1vQKOSZP29z4OJA327iSPq4e7pfPF8asl6Yjvzw_pUNWHLDrEttYbub2yHpGEDTyylmaYnUpW0mJXDDf/pub?gid=1357971635&single=true&output=csv"
    ).then(rawData => {
        // Clean data and parse odometer integers
        let logs = rawData
            .filter(d => d.Mileage && d.Year !== "1899")
            .map(d => ({
                mileage: +d.Mileage.replace(/,/g, "")
            }))
            .filter(d => d.mileage > 0)
            .sort((a, b) => a.mileage - b.mileage); // Ensure chronological order

        // Dynamically compute intervals between entries
        let tripDistances = [];
        for (let i = 1; i < logs.length; i++) {
            let diff = logs[i].mileage - logs[i].mileage - logs[i-1].mileage; 
            // Real logic: current row odometer minus the immediate previous row odometer
            let trip = logs[i].mileage - logs[i-1].mileage;
            
            // Sanity filter: Ignore gaps from missed entries (e.g., accidental giant gaps > 800 miles)
            if (trip > 0 && trip < 800) { 
                tripDistances.push(trip);
            }
        }

        // Set Up Layout Dimensions
        const margin = { top: 40, right: 30, bottom: 50, left: 50 };
        const width = 600 - margin.left - margin.right;
        const height = 400 - margin.top - margin.bottom;

        const svg = d3.select("#histogram")
            .append("svg")
            .attr("width", width + margin.left + margin.right)
            .attr("height", height + margin.top + margin.bottom)
            .append("g")
            .attr("transform", `translate(${margin.left},${margin.top})`);

        // Create X Scale 
        const xScale = d3.scaleLinear()
            .domain([0, d3.max(tripDistances) * 1.05]) // Leave room on right bound
            .range([0, width]);

        // Build Histogram Bin Generator
        // Generates discrete threshold bars (e.g., every 50 miles)
        const histogram = d3.histogram() // Note: Use d3.bin() if using D3 v6 or higher
            .value(d => d)
            .domain(xScale.domain())
            .thresholds(xScale.ticks(16));

        const bins = histogram(tripDistances);

        // Create Y Scale based on max bin frequency count
        const yScale = d3.scaleLinear()
            .domain([0, d3.max(bins, d => d.length) + 1])
            .range([height, 0]);

        // Render Layout Axes
        svg.append("g")
            .attr("transform", `translate(0,${height})`)
            .call(d3.axisBottom(xScale))
            .append("text")
            .attr("x", width / 2)
            .attr("y", 40)
            .attr("fill", "#000")
            .style("text-anchor", "middle")
            .style("font-weight", "bold")
            .text("Distance Between Fill-ups (Miles)");

        svg.append("g")
            .call(d3.axisLeft(yScale).ticks(d3.max(bins, d => d.length))) // Integer ticks only
            .append("text")
            .attr("transform", "rotate(-90)")
            .attr("x", -height / 2)
            .attr("y", -35)
            .attr("fill", "#000")
            .style("text-anchor", "middle")
            .style("font-weight", "bold")
            .text("Frequency (Number of Fill-ups)");

        // Draw Histogram Bars
        svg.selectAll(".bar")
            .data(bins)
            .enter()
            .append("rect")
            .attr("class", "bar")
            .attr("x", d => xScale(d.x0) + 1) // 1px space padding on left boundary
            .attr("y", d => yScale(d.length))
            .attr("width", d => Math.max(0, xScale(d.x1) - xScale(d.x0) - 1)) // Bar width minus gap padding
            .attr("height", d => height - yScale(d.length))
            .attr("fill", "#dddddd")
            .attr("stroke", "#777777")
            .attr("stroke-width", 1);

        // Static Labels over bars for absolute readability
        svg.selectAll(".bar-label")
            .data(bins)
            .enter()
            .append("text")
            .attr("class", "bar-label")
            .attr("x", d => xScale(d.x0) + (xScale(d.x1) - xScale(d.x0)) / 2)
            .attr("y", d => yScale(d.length) - 6)
            .attr("fill", "#333")
            .style("text-anchor", "middle")
            .style("font-size", "11px")
            .text(d => d.length > 0 ? d.length : ""); // Show nothing if count is zero

    }).catch(err => console.error("Error drawing error-bar chart:", err));
};

drawChart2();
</script>

<script>
const drawChart3 = async () => {

// --------------------------------
// Chart 3: Fuel price over time (Line Chart)
// --------------------------------
d3.csv(
    "https://docs.google.com/spreadsheets/d/e/2PACX-1vQKOSZP29z4OJA327iSPq4e7pfPF8asl6Yjvzw_pUNWHLDrEttYbub2yHpGEDTyylmaYnUpW0mJXDDf/pub?gid=1357971635&single=true&output=csv"
).then(rawData => {
    // Clean and transform raw data
    const parseDate = d3.timeParse("%Y-%m-%d");
    const cleanData = rawData
        // .filter(d => d.Price && !isNaN(d.Price))
        .map(d => ({
            date: parseDate(d.Date),
            price: 0 + d.Price.slice(1) // Remove leading $ sign and convert to number,
        }))
        .filter(d => d.price > 0);
    console.log(cleanData); 

    //  Set dimensions and margins
    const margin = { top: 30, right: 30, bottom: 50, left: 50 };
    const width = 600 - margin.left - margin.right;
    const height = 400 - margin.top - margin.bottom;

    // Append SVG container
    const svg = d3.select("#dieselprice")
        .append("svg")
        .attr("width", width + margin.left + margin.right)
        .attr("height", height + margin.top + margin.bottom)
        .append("g")
        .attr("transform", `translate(${margin.left},${margin.top})`);

    // Define Scales
    const xScale = d3.scaleTime()
        .domain(d3.extent(cleanData, d => d.date))
        .range([0, width]);

    const yScale = d3.scaleLinear()
        .domain([
        d3.min(cleanData, d => d.price) * 0.9,
        d3.max(cleanData, d => d.price) * 1.1
        ])
        .range([height, 0]);

    // Draw Y-Axis Gridlines first (so they sit behind the paths)
    svg.append("g")
        .attr("class", "grid")
        .call(d3.axisLeft(yScale)
            .tickSize(-width)
            .tickFormat("")
        );

    // Define generators with smooth curves (curveBasis provides a very smooth spline)

    const smoothLineGenerator = d3.line()
        .x(d => xScale(d.date))
        .y(d => yScale(d.price))
        .curve(d3.curveBasis);

    // Append the Stroke Line
    svg.append("path")
        .datum(cleanData)
        .attr("fill", "none")
        .attr("stroke", "#777")
        .attr("stroke-width", 2)
        .attr("d", smoothLineGenerator);

    // Add X & Y Axes
    svg.append("g")
        .attr("class", "axis")
        .attr("transform", `translate(0,${height})`)
        .call(d3.axisBottom(xScale).ticks(5).tickFormat(d3.timeFormat("%b %Y")));

    svg.append("g")
        .attr("class", "axis")
        .call(d3.axisLeft(yScale).tickFormat(d => `$${d.toFixed(2)}`));
    }).catch(err => console.error("Error drawing error-bar chart:", err));
};

drawChart3();
</script>
